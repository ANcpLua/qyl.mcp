from pathlib import Path
import re
root=Path.cwd()
s=Path('release/contract.md').read_text()
work=s.split('## Work-item mapping',1)[1].split('## Done-when mapping',1)[0]
done=s.split('## Done-when mapping',1)[1]
assert list(map(int,re.findall(r'^\| (\d+) \|',work,re.M)))==list(range(1,16))

assert '| 14 | Pending:' in work and '| 15 | Pending:' in work
assert 'permitted alternative' in done
print('Pending points 14/15 and point-8 release blocker: explicit')
links=0
for f in ['release/README.md','release/contract.md','MCP-CHECKPOINT.md']:
 path=root/f
 for target in re.findall(r'\]\(([^)]+)\)',path.read_text()):
  if '://' in target or target.startswith('#'): continue
  name,_,anchor=target.partition('#');dest=(path.parent/name).resolve()
  assert dest.is_file(),(f,target)
  if anchor and dest.suffix=='.md':
   headings=re.findall(r'^#+\s+(.+)$',dest.read_text(),re.M)
   slugs=[re.sub(r'[^\w\- ]','',h.lower()).replace(' ','-') for h in headings]
   assert anchor in slugs,(f,target)
  links+=1
print(f'Local document links and heading anchors: {links} resolve')
