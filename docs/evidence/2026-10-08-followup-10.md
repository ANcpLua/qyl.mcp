`validate.py`:

```python
import hashlib,json,re
from pathlib import Path
from zipfile import ZipFile
from jsonschema import Draft202012Validator
root=Path('submission/qyl')
for name in ['plugin','mcp']:
    schema=json.loads(Path('submission/schemas/'+name+'.schema.json').read_text())
    value=json.loads((root/(name+'.json')).read_text())
    assert value['$schema']==schema['$id']
    Draft202012Validator.check_schema(schema)
    Draft202012Validator(schema).validate(value)
    print(name+'.json: PASS against '+value['$schema'])
for rel in ['.claude-plugin/plugin.json','.mcp.json','README.md','LICENSE','skills/qyl-investigate/SKILL.md']:
    assert (root/rel).is_file(); print(rel+': present')
words=len(re.sub(r'```.*?```','',(root/'README.md').read_text(),flags=re.S).split())
assert words>=40; print('README words outside code blocks:',words)
plugin=json.loads((root/'plugin.json').read_text()); meta=plugin['extensions']['com.openai']
assert len(meta['review']['test_cases']['positive'])==5 and len(meta['review']['test_cases']['negative'])==3
assert len(json.loads((root/'mcp.json').read_text())['mcpServers'])==1
assert 'developerName' not in meta['interface'] and 'countries' not in meta['publication']
print('review cases: 5 positive, 3 negative; servers: 1; owner identity/countries unset')
archive=Path('submission/packages/qyl-openai-1.0.0-draft.zip')
with ZipFile(archive) as z:
    names=z.namelist()
    assert names==['LICENSE','README.md','assets/qyl-icon.png','mcp.json','plugin.json','skills/qyl-investigate/SKILL.md']
    for name in names: assert z.read(name)==(root/name).read_bytes()
    assert 'apps' not in plugin and 'apps' not in meta
    print('ZIP: six source-identical files; no .claude-plugin, .mcp.json, .app.json or apps binding')
print('ZIP sha256:',hashlib.sha256(archive.read_bytes()).hexdigest())
```
